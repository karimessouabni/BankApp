#!/usr/bin/env bash
set -euo pipefail

: "${IBM_API_KEY:?IBM_API_KEY non défini}"
BUCKET="${BUCKET:-bu002i009999}"

IAM_URL="https://private.iam.cloud.ibm.com/identity/token"
ENDPOINT="https://s3.direct.eu-fr2.cloud-object-storage.appdomain.cloud:443/${BUCKET}"

get_token() {
  curl -sS -k -X POST "$IAM_URL" \
    -H 'Accept: application/json' \
    -H 'Content-Type: application/x-www-form-urlencoded' \
    --data-urlencode "apikey=${IBM_API_KEY}" \
    --data-urlencode 'response_type=cloud_iam' \
    --data-urlencode 'grant_type=urn:ibm:params:oauth:grant-type:apikey' \
  | python3 -c 'import sys,json; print(json.load(sys.stdin)["access_token"])'
}

TOKEN="$(get_token)" || { echo "Échec IAM" >&2; exit 1; }
[[ -n "$TOKEN" ]] || { echo "Token vide" >&2; exit 1; }

# wrapper : cos <METHOD> <path/query> [args curl...]
cos() {
  local method="$1" path="$2"; shift 2
  curl -sS -k -X "$method" "${ENDPOINT}${path}" \
    -H "Authorization: Bearer ${TOKEN}" "$@"
}
