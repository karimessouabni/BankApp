#!/usr/bin/env bash
#
# Vide un bucket IBM COS (suppression de tous les objets).
#
# Usage:
#   export IBM_API_KEY=...            # ou --apikey
#   ./empty-bucket-script.sh --bucket bu002i007154 --dry-run
#   ./empty-bucket-script.sh --bucket bu002i007154
#
# Note: les multipart uploads incomplets ne sont PAS des objets et ne sortent
#       pas de ListObjects. Lancer multipart-cleaning.sh avant le delete du
#       bucket, sinon il repart en 409.
#
set -euo pipefail

# ---------------------------------------------------------------- config ----
IAM_URL="${IAM_URL:-https://private.iam.cloud.ibm.com/identity/token}"
COS_HOST="${COS_HOST:-s3.direct.eu-fr2.cloud-object-storage.appdomain.cloud:443}"
BUCKET="${BUCKET:-}"
API_KEY="${IBM_API_KEY:-}"
DRY_RUN=0
CURL_OPTS=(-sS -k)          # -k : chaîne TLS interne BNP

while [[ $# -gt 0 ]]; do
  case "$1" in
    --bucket)  BUCKET="$2"; shift 2 ;;
    --apikey)  API_KEY="$2"; shift 2 ;;
    --host)    COS_HOST="$2"; shift 2 ;;
    --dry-run) DRY_RUN=1; shift ;;
    -h|--help) sed -n '2,13p' "$0"; exit 0 ;;
    *) echo "Option inconnue : $1" >&2; exit 2 ;;
  esac
done

[[ -n "$BUCKET"  ]] || { echo "ERREUR: --bucket ou \$BUCKET requis" >&2; exit 2; }
[[ -n "$API_KEY" ]] || { echo "ERREUR: --apikey ou \$IBM_API_KEY requis" >&2; exit 2; }

ENDPOINT="https://${COS_HOST}/${BUCKET}"

# ------------------------------------------------------------------ IAM -----
TOKEN=""
TOKEN_EXP=0

refresh_token() {
  local json
  json=$(curl "${CURL_OPTS[@]}" -X POST "$IAM_URL" \
      -H 'Accept: application/json' \
      -H 'Content-Type: application/x-www-form-urlencoded' \
      --data-urlencode "apikey=${API_KEY}" \
      --data-urlencode 'response_type=cloud_iam' \
      --data-urlencode 'grant_type=urn:ibm:params:oauth:grant-type:apikey') \
    || { echo "ERREUR: appel IAM échoué" >&2; exit 1; }

  TOKEN=$(printf '%s' "$json" | python3 -c \
    'import sys,json; print(json.load(sys.stdin).get("access_token",""))') || true

  if [[ -z "$TOKEN" ]]; then
    echo "ERREUR: pas d'access_token dans la réponse IAM :" >&2
    printf '%s\n' "$json" | head -c 400 >&2; echo >&2
    exit 1
  fi

  local ttl
  ttl=$(printf '%s' "$json" | python3 -c \
    'import sys,json; print(json.load(sys.stdin).get("expires_in",3600))')
  TOKEN_EXP=$(( $(date +%s) + ttl - 300 ))   # marge de 5 min
  echo "Token IAM obtenu (valide ~$(( ttl / 60 )) min)."
}

ensure_token() {
  (( $(date +%s) < TOKEN_EXP )) || refresh_token
}

# ---------------------------------------------------------------- helpers ---
HTTP_CODE=""
cos() {
  local method="$1" path="$2"; shift 2
  ensure_token
  local out
  out=$(curl "${CURL_OPTS[@]}" -w '\n%{http_code}' -X "$method" \
        --url "${ENDPOINT}${path}" \
        -H "Authorization: Bearer ${TOKEN}" "$@") \
    || { echo "ERREUR: curl a échoué (réseau/proxy) sur ${method} ${path}" >&2; return 1; }
  HTTP_CODE="${out##*$'\n'}"
  printf '%s' "${out%$'\n'*}"
}

urlencode_key() {
  # garde les "/" ; python3 gère l'UTF-8 là où printf de bash foire >127
  python3 -c "import sys, urllib.parse; print(urllib.parse.quote(sys.argv[1], safe='/'))" "$1"
}

xp() { xmllint --xpath "$1" - 2>/dev/null || true; }

# ------------------------------------------------------------------ main ----
echo "Bucket   : $BUCKET"
echo "Endpoint : $ENDPOINT"
(( DRY_RUN )) && echo "Mode     : DRY-RUN (aucune suppression réelle)"
echo

MARKER=""
TOTAL=0
DELETED=0
FAILED=0
PAGE=0

while :; do
  PAGE=$(( PAGE + 1 ))
  QUERY="?max-keys=1000"
  [[ -n "$MARKER" ]] && QUERY+="&marker=$(urlencode_key "$MARKER")"

  XML=$(cos GET "$QUERY")
  if [[ "$HTTP_CODE" != "200" ]]; then
    echo "ERREUR: ListObjects a renvoyé HTTP $HTTP_CODE" >&2
    printf '%s\n' "$XML" | head -c 600 >&2; echo >&2
    exit 1
  fi

  COUNT=$(printf '%s' "$XML" | xp 'count(//*[local-name()="Contents"])')
  COUNT="${COUNT:-0}"
  echo "Page $PAGE : $COUNT objet(s)."

  if (( COUNT == 0 )); then
    (( PAGE == 1 )) && echo "Bucket déjà vide."
    break
  fi

  LAST_KEY=""
  for (( i=1; i<=COUNT; i++ )); do
    KEY=$(printf '%s' "$XML" | xp "string((//*[local-name()='Contents'])[$i]/*[local-name()='Key']/text())")

    if [[ -z "$KEY" ]]; then
      echo "  /!\\ Objet #$i ignoré (key vide)"
      FAILED=$(( FAILED + 1 ))
      continue
    fi
    LAST_KEY="$KEY"

    TOTAL=$(( TOTAL + 1 ))
    ENCODED_KEY=$(urlencode_key "$KEY")
    echo "  [$TOTAL] $KEY"

    if (( DRY_RUN )); then
      echo "        (dry-run) DELETE ${ENDPOINT}/${ENCODED_KEY}"
      continue
    fi

    BODY=$(cos DELETE "/${ENCODED_KEY}") || { FAILED=$(( FAILED + 1 )); continue; }
    if [[ "$HTTP_CODE" == "204" || "$HTTP_CODE" == "200" ]]; then
      echo "        -> HTTP $HTTP_CODE OK"
      DELETED=$(( DELETED + 1 ))
    else
      echo "        -> HTTP $HTTP_CODE ECHEC"
      [[ -n "$BODY" ]] && printf '           %s\n' "$(printf '%s' "$BODY" | head -c 300)"
      FAILED=$(( FAILED + 1 ))
    fi
  done

  TRUNCATED=$(printf '%s' "$XML" | xp 'string(//*[local-name()="IsTruncated"]/text())')
  if [[ "$TRUNCATED" == "true" ]]; then
    if (( DRY_RUN )); then
      # rien n'est supprimé : on avance avec le marker
      NEXT=$(printf '%s' "$XML" | xp 'string(//*[local-name()="NextMarker"]/text())')
      MARKER="${NEXT:-$LAST_KEY}"
      [[ -n "$MARKER" ]] || break
    else
      # les objets ont été supprimés : on repart du début du bucket
      MARKER=""
    fi
  else
    break
  fi
done

echo
echo "Résumé : $TOTAL trouvé(s), $DELETED supprimé(s), $FAILED en échec."
(( FAILED == 0 ))
