source .venv/bin/activate
export ALEMBIC_ENV=preprod
python - <<'EOF'
import os
from dotenv import load_dotenv
load_dotenv(".env.preprod")
from sqlalchemy import create_engine, text
e = create_engine(os.environ["DATABASE_URL"])
with e.connect() as c:
    print(c.execute(text("select version(), current_database(), current_user")).one())
EOF
